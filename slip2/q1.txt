SLIP 2 - Q.1
Question: Demonstrate constructor injection in Spring by creating a Course class containing course name and a List of subjects. Configure the bean using XML.

Files needed: pom.xml (add these dependencies), Course.java, beans.xml, MainApp.java

====================  pom.xml (add these dependencies)  ====================
<dependencies>
    <!-- Spring Core + Context -->
    <dependency>
        <groupId>org.springframework</groupId>
        <artifactId>spring-context</artifactId>
        <version>5.3.39</version>
    </dependency>
</dependencies>

====================  Course.java  ====================
package com.example;

import java.util.List;

public class Course {
    private String courseName;
    private List<String> subjects;

    // Constructor Injection
    public Course(String courseName, List<String> subjects) {
        this.courseName = courseName;
        this.subjects = subjects;
    }

    public void display() {
        System.out.println("Course Name : " + courseName);
        System.out.println("Subjects    : " + subjects);
    }
}

====================  beans.xml  ====================
<?xml version="1.0" encoding="UTF-8"?>
<beans xmlns="http://www.springframework.org/schema/beans"
       xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
       xsi:schemaLocation="http://www.springframework.org/schema/beans
       https://www.springframework.org/schema/beans/spring-beans.xsd">
    <bean id="course" class="com.example.Course">
        <constructor-arg value="M.Sc Computer Science"/>
        <constructor-arg>
            <list>
                <value>Advanced Java</value>
                <value>Machine Learning</value>
                <value>Agile Software Development</value>
            </list>
        </constructor-arg>
    </bean>
</beans>

====================  MainApp.java  ====================
package com.example;

import org.springframework.context.ApplicationContext;
import org.springframework.context.support.ClassPathXmlApplicationContext;

public class MainApp {
    public static void main(String[] args) {
        ApplicationContext ctx = new ClassPathXmlApplicationContext("beans.xml");
        Course c = ctx.getBean("course", Course.class);
        c.display();
    }
}
